```typescript
import type { OperationParameters, OperationType, OperationStep, PipelinePlan } from '../types/api';
import type { ParameterState, PipelineStepDraft, PipelineStepTemplate } from '../types/image';
import { defaultParameterState } from '../config/imageOperations';

export function createId(prefix: string): string {
  return `${prefix}-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
}

export function createPipelineStep(
  operation: OperationType,
  overrides: Partial<ParameterState> = {},
): PipelineStepDraft {
  return {
    id: createId('step'),
    operation,
    parameters: { ...defaultParameterState, ...overrides },
  };
}

export function clonePipelineTemplates(steps: PipelineStepTemplate[]): PipelineStepDraft[] {
  return steps.map((step) => createPipelineStep(step.operation, step.parameters));
}

export function toPipelineTemplates(steps: PipelineStepDraft[]): PipelineStepTemplate[] {
  return steps.map((step) => ({
    operation: step.operation,
    parameters: { ...step.parameters },
  }));
}

const parameterBuilders: Partial<Record<OperationType, (p: ParameterState) => OperationParameters>> = {
  blur:             (p) => ({ blur_strength: p.blurStrength }),
  brightness:       (p) => ({ brightness: p.brightness }),
  contrast:         (p) => ({ contrast: p.contrast }),
  sharpen:          (p) => ({ sharpen_strength: p.sharpenStrength }),
  denoise:          (p) => ({ denoise_strength: p.denoiseStrength }),
  object_detection: (p) => ({ detection_threshold: p.detectionThreshold / 100 }),
};

export function buildOperationStep(step: PipelineStepDraft): OperationStep {
  const builder = parameterBuilders[step.operation];
  return {
    operation: step.operation,
    parameters: builder ? builder(step.parameters) : undefined,
  };
}

export function pipelinePlanStepToTemplate(step: OperationStep): PipelineStepTemplate {
  return {
    operation: step.operation,
    parameters: {
      ...defaultParameterState,
      blurStrength: step.parameters?.blur_strength ?? defaultParameterState.blurStrength,
      brightness: step.parameters?.brightness ?? defaultParameterState.brightness,
      contrast: step.parameters?.contrast ?? defaultParameterState.contrast,
      sharpenStrength: step.parameters?.sharpen_strength ?? defaultParameterState.sharpenStrength,
      denoiseStrength: step.parameters?.denoise_strength ?? defaultParameterState.denoiseStrength,
      detectionThreshold: Math.round(
        (step.parameters?.detection_threshold ?? defaultParameterState.detectionThreshold / 100) * 100,
      ),
    },
  };
}

export function pipelinePlanToTemplates(plan: PipelinePlan): PipelineStepTemplate[] {
  return plan.steps.map(pipelinePlanStepToTemplate);
}

export function movePipelineStep(
  steps: PipelineStepDraft[],
  stepId: string,
  direction: 'up' | 'down',
): PipelineStepDraft[] {
  const currentIndex = steps.findIndex((step) => step.id === stepId);

  if (currentIndex === -1) {
    return steps;
  }

  const targetIndex = direction === 'up' ? currentIndex - 1 : currentIndex + 1;

  if (targetIndex < 0 || targetIndex >= steps.length) {
    return steps;
  }

  const reordered = [...steps];
  const [movedStep] = reordered.splice(currentIndex, 1);
  reordered.splice(targetIndex, 0, movedStep);
  return reordered;
}
```

  context.fillStyle = background;
  context.fillRect(0, 0, canvas.width, canvas.height);

  context.fillRect(120, 110, 390, 540);

  context.fillStyle = '#f8fafc';

  context.arc(860, 330, 120, 0, Math.PI * 2);

  context.strokeRect(780, 455, 240, 110);

  context.fillStyle = '#ffffff';
  context.font = 'bold 44px sans-serif';
  context.fillText('SAMPLE', 798, 520);

  const blob = await new Promise<Blob>((resolve, reject) => {
    canvas.toBlob((nextBlob) => {
      if (!nextBlob) {
        reject(new Error('デモ画像の生成に失敗しました。'));
        return;
      }
      resolve(nextBlob);
    }, 'image/png');
  });

  return new File([blob], 'smart-image-studio-demo.png', { type: 'image/png' });

export function useFile() {
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  useEffect(() => {
    return () => {
      if (previewUrl) URL.revokeObjectURL(previewUrl);
    };
  }, [previewUrl]);

  function selectFile(file: File | null) {
    if (previewUrl) URL.revokeObjectURL(previewUrl);

    if (!file) {
      setSelectedFile(null);
      setPreviewUrl(null);
      setErrorMessage(null);
      return;

    const validationError = validateImageFile(file);

    if (validationError) {
      setSelectedFile(null);
      setPreviewUrl(null);
      setErrorMessage(validationError);
      return;
    }

    setSelectedFile(file);
    setPreviewUrl(URL.createObjectURL(file));
    setErrorMessage(null);
  }

  async function loadDemoImage() {
    try {
      const demo = await createDemoImageFile();
      selectFile(demo);
    } catch (error) {
      setErrorMessage(toErrorMessage(error, 'デモ画像の読み込みに失敗しました。'));
    }
  }

  return { selectedFile, previewUrl, errorMessage, selectFile, loadDemoImage };

```typescript
import { useState } from 'react';
import type { OperationType, PipelinePlan } from '../types/api';
import type { ParameterKey, PipelinePreset, PipelineStepDraft } from '../types/image';
import {
  clonePipelineTemplates,
  createId,
  createPipelineStep,
  movePipelineStep,
  pipelinePlanToTemplates,
  toPipelineTemplates,
} from '../utils/pipeline';
import {
  deletePipelinePreset,
  readSavedPipelineDraft,
  readSavedPipelinePresets,
  savePipelineDraft,
  savePipelinePreset,
} from '../utils/pipelineStorage';

function toErrorMessage(error: unknown, fallback: string): string {
  if (error instanceof Error && error.message.trim() !== '') return error.message;
  return fallback;
}

export function usePipeline() {
  const initialDraft = readSavedPipelineDraft();

  const [steps, setSteps] = useState<PipelineStepDraft[]>(
    initialDraft ? clonePipelineTemplates(initialDraft.steps) : [],
  );
  const [savedPresets, setSavedPresets] = useState<PipelinePreset[]>(() => readSavedPipelinePresets());
  const [presetName, setPresetName] = useState('');
  const [savedDraftAt, setSavedDraftAt] = useState<string | null>(initialDraft?.updatedAt ?? null);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  function addStep(operation: OperationType) {
    setSteps((current) => [...current, createPipelineStep(operation)]);
    setErrorMessage(null);
  }

  function removeStep(stepId: string) {
    setSteps((current) => current.filter((s) => s.id !== stepId));
    setErrorMessage(null);
  }

  function moveStep(stepId: string, direction: 'up' | 'down') {
    setSteps((current) => movePipelineStep(current, stepId, direction));
  }

  function updateStepParameter(stepId: string, key: ParameterKey, value: number) {
    setSteps((current) =>
      current.map((s) =>
        s.id === stepId ? { ...s, parameters: { ...s.parameters, [key]: value } } : s,
      ),
    );
  }

  function applyPreset(preset: PipelinePreset) {
    setSteps(clonePipelineTemplates(preset.steps));
    setPresetName(preset.name);
    setErrorMessage(null);
  }

  function applyPlan(plan: PipelinePlan) {
    setSteps(clonePipelineTemplates(pipelinePlanToTemplates(plan)));
    setErrorMessage(null);
  }

  function saveDraft() {
    if (steps.length === 0) {
      setErrorMessage('保存する前に 1 つ以上のステップを追加してください。');
      return;
    }
    try {
      const draft = savePipelineDraft(steps);
      setSavedDraftAt(draft.updatedAt);
      setErrorMessage(null);
    } catch (error) {
      setErrorMessage(toErrorMessage(error, 'パイプラインの保存に失敗しました。'));
    }
  }

  function loadDraft() {
    try {
      const draft = readSavedPipelineDraft();
      if (!draft) {
        setErrorMessage('保存済みパイプラインがありません。');
        return;
      }
      setSteps(clonePipelineTemplates(draft.steps));
      setSavedDraftAt(draft.updatedAt);
      setErrorMessage(null);
    } catch (error) {
      setErrorMessage(toErrorMessage(error, '保存済みパイプラインの読み込みに失敗しました。'));
    }
  }

  function savePreset() {
    if (presetName.trim() === '') {
      setErrorMessage('プリセット名を入力してください。');
      return;
    }
    if (steps.length === 0) {
      setErrorMessage('保存する前に 1 つ以上のステップを追加してください。');
      return;
    }
    const preset: PipelinePreset = {
      id: createId('preset'),
      name: presetName.trim(),
      description: `${steps.length} ステップのカスタムプリセット`,
      steps: toPipelineTemplates(steps),
    };
    try {
      setSavedPresets(savePipelinePreset(preset));
      setPresetName('');
      setErrorMessage(null);
    } catch (error) {
      setErrorMessage(toErrorMessage(error, 'プリセットの保存に失敗しました。'));
    }
  }

  function deletePreset(presetId: string) {
    try {
      setSavedPresets(deletePipelinePreset(presetId));
      setErrorMessage(null);
    } catch (error) {
      setErrorMessage(toErrorMessage(error, 'プリセットの削除に失敗しました。'));
    }
  }

  return {
    steps,
    savedPresets,
    presetName,
    setPresetName,
    savedDraftAt,
    errorMessage,
    addStep,
    removeStep,
    moveStep,
    updateStepParameter,
    applyPreset,
    applyPlan,
    saveDraft,
    loadDraft,
    savePreset,
    deletePreset,
  };
}
```

import { useState } from 'react';
import type { PipelineInterpreterMode, PipelinePlan } from '../types/api';
import type { PlannerHistoryEntry } from '../types/image';
import { interpretPipeline } from '../api/pipelineApi';
import { createId } from '../utils/pipeline';

function toErrorMessage(error: unknown, fallback: string): string {
  if (error instanceof Error && error.message.trim() !== '') return error.message;
  return fallback;
}

export function usePlanner() {
  const [instruction, setInstruction] = useState('');
  const [mode, setMode] = useState<PipelineInterpreterMode>('rule_based');
  const [plan, setPlan] = useState<PipelinePlan | null>(null);
  const [isInterpreting, setIsInterpreting] = useState(false);
  const [history, setHistory] = useState<PlannerHistoryEntry[]>([]);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  async function generatePlan() {
    if (instruction.trim() === '') {
      setErrorMessage('自然言語の指示を入力してください。');
      return;
    }

    setIsInterpreting(true);

    try {
      const response = await interpretPipeline({ instruction: instruction.trim(), mode });

      setPlan(response.plan);
      setHistory((current) =>
        [
          {
            id: createId('plan'),
            createdAt: new Date().toISOString(),
            mode: response.plan.interpreter,
            instruction: instruction.trim(),
            plan: response.plan,
          },
          ...current,
        ].slice(0, 6),
      );
      setErrorMessage(response.plan.steps.length === 0 ? '解釈結果に実行可能な step がありません。' : null);
    } catch (error) {
      setPlan(null);
      setErrorMessage(toErrorMessage(error, '自然文の解釈に失敗しました。'));
    } finally {
      setIsInterpreting(false);
    }
  }

  function selectHistory(entryId: string) {
    const entry = history.find((e) => e.id === entryId);
    if (!entry) return;
    setInstruction(entry.instruction);
    setMode(entry.mode);
    setPlan(entry.plan);
  }

  function selectExample(value: string) {
    setInstruction(value);
    setPlan(null);
  }

  return {
    instruction,
    setInstruction,
    mode,
    setMode,
    plan,
    isInterpreting,
    history,
    errorMessage,
    generatePlan,
    selectHistory,
    selectExample,
  };
}

```typescript
import { useEffect, useRef, useState } from 'react';
import type { ImageProcessJob, OperationStep, PipelinePlan, ProcessImageResponse } from '../types/api';
import { createProcessJob, fetchProcessJob, fetchProcessJobs } from '../api/imageApi';
import { processPipelineImage } from '../api/pipelineApi';
import { saveJobNotice } from '../utils/jobNoticeStore';
import { saveLatestProcessResult } from '../utils/processResultStore';

interface UseJobTrackingOptions {
  onCompleted?: (result: ProcessImageResponse) => void;
}

function toErrorMessage(error: unknown, fallback: string): string {
  if (error instanceof Error && error.message.trim() !== '') return error.message;
  return fallback;
}

export function useJobTracking({ onCompleted }: UseJobTrackingOptions = {}) {
  const onCompletedRef = useRef(onCompleted);
  const handledJobsRef = useRef<Set<string>>(new Set());

  const [activeJob, setActiveJob] = useState<ImageProcessJob | null>(null);
  const [jobHistory, setJobHistory] = useState<ImageProcessJob[]>([]);
  const [isProcessing, setIsProcessing] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  useEffect(() => {
    onCompletedRef.current = onCompleted;
  }, [onCompleted]);

  useEffect(() => {
    let cancelled = false;

    async function load() {
      try {
        const jobs = await fetchProcessJobs();
        if (cancelled) return;
        setJobHistory(jobs);
        const active = jobs.find((j) => j.status === 'queued' || j.status === 'running') ?? null;
        setActiveJob(active);
        setIsProcessing(active !== null);
      } catch {
        if (!cancelled) setErrorMessage('単体処理のジョブ履歴取得に失敗しました。');
      }
    }

    void load();
    return () => { cancelled = true; };
  }, []);

  useEffect(() => {
    const jobId = activeJob?.job_id;
    const jobStatus = activeJob?.status;

    if (!jobId || !jobStatus || !['queued', 'running'].includes(jobStatus)) return;

    let cancelled = false;

    async function poll() {
      try {
        const [job, jobs] = await Promise.all([fetchProcessJob(jobId!), fetchProcessJobs()]);

        if (cancelled) return;

        setActiveJob(job);
        setJobHistory(jobs);
        setIsProcessing(job.status === 'queued' || job.status === 'running');

        if (handledJobsRef.current.has(job.job_id)) return;

        if (job.status === 'completed') {
          handledJobsRef.current.add(job.job_id);

          if (!job.result) {
            setIsProcessing(false);
            setErrorMessage('ジョブは完了しましたが、結果を取得できませんでした。');
            return;
          }

          saveLatestProcessResult(job.result);
          saveJobNotice({ tone: 'success', message: `ジョブ ${job.job_id.slice(0, 8)} が完了しました。` });
          setIsProcessing(false);
          setErrorMessage(null);
          onCompletedRef.current?.(job.result);
        }

        if (job.status === 'failed') {
          handledJobsRef.current.add(job.job_id);
          setIsProcessing(false);
          setErrorMessage(job.error_message ?? '画像処理ジョブが失敗しました。');
        }
      } catch (error) {
        if (!cancelled) {
          setIsProcessing(false);
          setErrorMessage(toErrorMessage(error, 'ジョブ状態の取得に失敗しました。'));
        }
      }
    }

    void poll();
    const id = window.setInterval(() => { void poll(); }, 1500);
    return () => { cancelled = true; window.clearInterval(id); };
  }, [activeJob?.job_id, activeJob?.status]);

  async function submitJob(file: File, operations: OperationStep[]) {
    setIsProcessing(true);
    setErrorMessage(null);

    try {
      const job = await createProcessJob({ file, operations });
      setActiveJob(job);
      setJobHistory((current) => [job, ...current.filter((j) => j.job_id !== job.job_id)]);
    } catch (error) {
      setIsProcessing(false);
      setErrorMessage(toErrorMessage(error, 'ジョブの作成に失敗しました。'));
    }
  }

  async function submitPipelinePlan(file: File, plan: PipelinePlan) {
    setIsProcessing(true);
    setErrorMessage(null);

    try {
      const result = await processPipelineImage({ file, plan });
      saveLatestProcessResult(result);
      setIsProcessing(false);
      setErrorMessage(null);
      onCompletedRef.current?.(result);
    } catch (error) {
      setIsProcessing(false);
      setErrorMessage(toErrorMessage(error, 'PipelinePlan の実行に失敗しました。'));
    }
  }

  function openJobResult(job: ImageProcessJob) {
    if (!job.result) {
      setErrorMessage('このジョブには表示できる結果がありません。');
      return;
    }
    saveLatestProcessResult(job.result);
    onCompletedRef.current?.(job.result);
  }

  return {
    activeJob,
    jobHistory,
    isProcessing,
    errorMessage,
    submitJob,
    submitPipelinePlan,
    openJobResult,
  };
}
```
